<body>
    <div class="chat-container">
        <div class="chat-header">Умный чат-бот</div>
        <div class="messages" id="messages"></div>
        <div class="input-container">
            <input type="text" id="userInput" placeholder="Введите сообщение..." autocomplete="off">
            <button id="sendButton" onclick="sendMessage()">Отправить</button>
        </div>
    </div>

    <script>
        const messages = document.getElementById('messages');
        const userInput = document.getElementById('userInput');
        const sendButton = document.getElementById('sendButton');
        
        const UIState = {
            IDLE: 'idle',
            TYPING: 'typing',
            SENDING: 'sending',
            ERROR: 'error'
        };
        
        let currentState = UIState.IDLE;
        
        let socket;
        function connectWebSocket() {
            console.log('Устанавливаем соединение с сервером...');

            setTimeout(() => {
                addSystemMessage('Соединение установлено');
            }, 1000);
        }
        
        function init() {
            connectWebSocket();
            setupEventListeners();
            addSystemMessage('Добро пожаловать в чат!');
        }
        
        function setupEventListeners() {
            userInput.addEventListener('keypress', (e) => {
                if (e.key === 'Enter') {
                    sendMessage();
                }
            });
        }
        
        function setState(newState) {
            currentState = newState;
            updateUI();
        }
        
        function updateUI() {
            sendButton.disabled = currentState === UIState.SENDING;
            userInput.disabled = currentState === UIState.SENDING;
            
            if (currentState === UIState.SENDING) {
                sendButton.textContent = 'Отправка...';
            } else {
                sendButton.textContent = 'Отправить';
            }
        }
        
        function sendMessage() {
            const text = userInput.value.trim();
            
            if (!text) {
                showError('Введите сообщение');
                return;
            }
            
            if (currentState === UIState.SENDING) return;
            
            setState(UIState.SENDING);
            addMessage(text, true);
            userInput.value = '';
            
            setTimeout(() => {
                receiveMessage('Это ответ от бота на ваше сообщение: "' + text + '"');
                setState(UIState.IDLE);
            }, 1000 + Math.random() * 2000);
        }
        
        function receiveMessage(text) {
            showTypingIndicator();
            
            setTimeout(() => {
                hideTypingIndicator();
                addMessage(text, false);
            }, 500 + Math.random() * 1500);
        }
        
        function addMessage(text, isUser) {
            const time = new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
            
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${isUser ? 'user-message' : 'bot-message'}`;
            messageDiv.innerHTML = `
                <div class="message-content">${formatMessage(text)}</div>
                <div class="message-time">${time}</div>
            `;
            
            messages.appendChild(messageDiv);
            scrollToBottom();
        }
        
        function addSystemMessage(text) {
            const messageDiv = document.createElement('div');
            messageDiv.className = 'system-message';
            messageDiv.textContent = text;
            messages.appendChild(messageDiv);
            scrollToBottom();
        }
        
        function formatMessage(text) {
            return text
                .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
                .replace(/\_(.*?)\_/g, '<em>$1</em>')
                .replace(/`(.*?)`/g, '<code>$1</code>');
        }
        
        function showTypingIndicator() {
            const typingDiv = document.createElement('div');
            typingDiv.className = 'typing-indicator';
            typingDiv.id = 'typingIndicator';
            typingDiv.innerHTML = `
                <span></span>
                <span></span>
                <span></span>
            `;
            messages.appendChild(typingDiv);
            scrollToBottom();
        }
        
        function hideTypingIndicator() {
            const typing = document.getElementById('typingIndicator');
            if (typing) typing.remove();
        }
        
        function showError(message) {
            const errorDiv = document.createElement('div');
            errorDiv.className = 'error-message';
            errorDiv.textContent = message;
            messages.appendChild(errorDiv);
            scrollToBottom();
            
            setTimeout(() => {
                errorDiv.remove();
            }, 3000);
        }
        
        function scrollToBottom() {
            messages.scrollTop = messages.scrollHeight;
        }
        
        window.onload = init;
    </script>
